cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(pico1_controller C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()

add_executable(pico1_controller src/main.c src/tokenizer.c)
target_include_directories(pico1_controller PRIVATE include)
target_link_libraries(pico1_controller pico_stdlib hardware_uart hardware_spi)
pico_enable_stdio_usb(pico1_controller 1)
pico_enable_stdio_uart(pico1_controller 0)
pico_add_extra_outputs(pico1_controller)
